def expand(value):
 result=[]
 for token in value.split(','):
  token=token.strip()
  m=re.fullmatch(r'(F|R2-)(\d+)([abc]?)(?:–(?:F|R2-)(\d+))?',token)
  if not m: return []
  prefix,first,suffix,last=m.groups()
  result += [prefix+str(n)+suffix for n in range(int(first),int(last or first)+1)]
 return result
