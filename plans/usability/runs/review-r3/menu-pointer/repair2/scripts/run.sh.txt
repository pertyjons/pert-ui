#!/usr/bin/env bash
# Usage: run.sh <label> <command...>
# Runs the command in the menu worktree with the menu target dir, logging the
# command, source hashes and mtimes before and after, the full output and the
# exit status to runs/NN-<label>.log. Never overwrites an existing log.
set -u
E=/home/per/pert-ui-usability/r3-menu-implementation2-evidence
W=/home/per/pert-ui-usability/r3-menu-worktree
export CARGO_TARGET_DIR=/home/per/pert-ui-usability/r3-menu-implementation-target
label=$1; shift
n=$(ls "$E/runs" | wc -l); n=$(printf '%02d' $((n + 1)))
log="$E/runs/$n-$label.log"
[ -e "$log" ] && { echo "refusing to overwrite $log"; exit 2; }
files="crates/pert-ui-runtime/src/select.rs crates/pert-ui-runtime/src/menu_bar.rs crates/pert-ui-runtime/src/menu.rs crates/pert-ui-runtime/src/activation.rs crates/pert-ui/tests/surfaces/main.rs crates/pert-ui/tests/surfaces/menu_pointer_ownership.rs"
cd "$W" || exit 2
{
  echo "# started $(date -Is)"
  echo "# cwd $W  CARGO_TARGET_DIR=$CARGO_TARGET_DIR"
  printf '# command:'; printf ' %q' "$@"; echo
  echo "# HEAD $(git rev-parse HEAD)"
  echo "# git status --porcelain:"; git status --porcelain
  echo "# hashes before:"; for f in $files; do [ -e "$f" ] && sha256sum "$f"; done
  echo "# mtimes before:"; for f in $files; do [ -e "$f" ] && stat -c '%.9Y %n' "$f"; done
  echo "# ---- output ----"
} > "$log"
"$@" >> "$log" 2>&1
status=$?
{
  echo "# ---- end output ----"
  echo "# hashes after:"; for f in $files; do [ -e "$f" ] && sha256sum "$f"; done
  echo "# finished $(date -Is)"
  echo "# exit_status $status"
} >> "$log"
echo "$log exit_status=$status"
exit $status
