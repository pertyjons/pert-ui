from pathlib import Path
import datetime
import json
import os
import subprocess

root = Path('/home/per/github/pert-ui')
output = Path('/home/per/pert-ui-usability/r3-disabled-button-gates')
output.mkdir(exist_ok=False)
commands = [
 ('format', ['cargo', 'fmt', '--all', '--check'], {}),
 ('clippy', ['cargo', 'clippy', '--locked', '--workspace', '--all-targets', '--', '-D', 'warnings'], {}),
 ('workspace-tests', ['cargo', 'test', '--locked', '--workspace'], {}),
 *[(name, ['cargo', 'run', '--locked', '-p', 'pert-ui', '--features', 'png,svg,cpu', '--example', name, *args], {})
   for name, args in [('live_mixer', []), ('dynamic_documents', []), ('diagram_editor', []), ('log_explorer', ['--', '--lines', '10000']), ('inventory', ['--', '--items', '2000'])]],
 ('doctests', ['cargo', 'test', '--locked', '--workspace', '--doc'], {}),
 ('rustdoc', ['cargo', 'doc', '--locked', '--workspace', '--no-deps'], {'RUSTDOCFLAGS': '-D warnings'}),
 ('diagnostics', ['cargo', 'test', '--locked', '-p', 'pert-ui-diagnostics', '--', '--ignored'], {}),
]
# Capture exact source bytes independent of later record-only updates.
import hashlib
files = subprocess.check_output(['git','ls-files','-co','--exclude-standard','-z'], cwd=root).split(b'\0')
source = {}
for raw in files:
 if not raw: continue
 rel = raw.decode()
 if rel.startswith('plans/') and not rel.startswith('plans/usability/task4-app/'): continue
 path = root / rel
 if path.is_file(): source[rel] = hashlib.sha256(path.read_bytes()).hexdigest()
(output / 'source-manifest.json').write_text(json.dumps(source, indent=2) + '\n')
records = []
for name, command, additions in commands:
 print('GATE START', name, flush=True)
 started = datetime.datetime.now(datetime.timezone.utc).isoformat()
 with (output / (name + '.log')).open('x') as log:
  log.write('$ ' + ' '.join(command) + '\n'); log.flush()
  result = subprocess.run(command, cwd=root, env=dict(os.environ, **additions), stdout=log, stderr=subprocess.STDOUT)
 records.append({'name': name, 'command': command, 'environment': additions, 'started': started,
                 'ended': datetime.datetime.now(datetime.timezone.utc).isoformat(), 'exit_status': result.returncode})
 (output / 'outcomes.json').write_text(json.dumps(records, indent=2) + '\n')
 print('GATE END', name, result.returncode, flush=True)
 if result.returncode:
  raise SystemExit(result.returncode)

for rel, digest in source.items():
 assert hashlib.sha256((root / rel).read_bytes()).hexdigest() == digest, rel
(output / 'source-unchanged.txt').write_text('All captured source bytes unchanged during gates.\n')
