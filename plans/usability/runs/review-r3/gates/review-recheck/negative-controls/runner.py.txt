from pathlib import Path
import hashlib
import json
import subprocess

root=Path('/home/per/github/pert-ui')
out=Path('/home/per/pert-ui-usability/r3-review-negative-controls')
out.mkdir(exist_ok=False)
experiments=[
 ('metadata-layout', 'crates/pert-ui-runtime/src/runtime/tree.rs',
  '            if !layout_changed {\n                continue;\n            }',
  '            if !layout_changed {\n                if let Some(parent) = entry.node.logical_parent() { parent.invalidate(Invalidation::Layout); }\n                continue;\n            }',
  ['cargo','test','--locked','-p','pert-ui','--test','foundation','names_replace_and_remove_without_remounting_or_scheduling_layout','--','--exact'],
  'was invalidated'),
 ('native-tab-no-input', 'apps/gallery/src/interaction_check.rs',
  '                driver.click_item(&tabs, 1).map_err(failed)',
  '                Ok(())',
  ['cargo','test','--locked','-p','pert-ui-gallery','cli::the_interaction_steps_pass_headlessly','--','--exact'],
  'Controls category title was not selected'),
]
records=[]
for name,file,old,new,command,expected in experiments:
 path=root/file
 original=path.read_bytes();text=original.decode()
 assert text.count(old)==1,(name,text.count(old))
 # The foundation test is a module in the area binary, so --exact names it fully.
 if name=='metadata-layout':command[7]='exposed_parts::names_replace_and_remove_without_remounting_or_scheduling_layout'
 mutated=text.replace(old,new).encode()
 try:
  path.write_bytes(mutated)
  with (out/(name+'.log')).open('x') as log:
   result=subprocess.run(command,cwd=root,stdout=log,stderr=subprocess.STDOUT)
 finally:
  path.write_bytes(original)
  assert path.read_bytes()==original,'Restoration failed'
 output=(out/(name+'.log')).read_text()
 record={'name':name,'file':file,'original_sha256':hashlib.sha256(original).hexdigest(),
         'mutated_sha256':hashlib.sha256(mutated).hexdigest(),'restored_sha256':hashlib.sha256(path.read_bytes()).hexdigest(),
         'command':command,'exit_status':result.returncode,'expected_failure_present':expected in output}
 records.append(record)
 (out/'outcomes.json').write_text(json.dumps(records,indent=2)+'\n')
 print(name,'exit',result.returncode,'expected message',expected in output,'original bytes restored',flush=True)
 assert result.returncode==101 and expected in output,'Negative control did not establish detection; inspect before proceeding'
