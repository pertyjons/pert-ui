#!/bin/sh
# Runs the probe against one build: baseline (the approved-tab tree) or
# implementation (the worktree). Output: $E/probe/out/<build>-<label>.out
E=/home/per/pert-ui-usability/r3-whitespace-implementation-evidence
build=$1; label=$2
if [ "$build" = baseline ]; then
  target=/home/per/pert-ui-usability/r3-whitespace-baseline-target; tree=$E/baseline-tree
else
  target=/home/per/pert-ui-usability/r3-whitespace-implementation-target; tree=/home/per/pert-ui-usability/r3-whitespace-worktree
fi
mkdir -p "$E/probe/out"
out="$E/probe/out/$build-$label.out"
{
  echo "started_at_utc: $(date -u +%FT%TZ)"
  echo "\$ cd $E/probe/$build && RUSTUP_TOOLCHAIN=1.99.1 CARGO_TARGET_DIR=$target cargo run --offline -q --release"
  echo "probe source: $(sha256sum "$E/probe/src/main.rs")"
  echo "manifest: $(sha256sum "$E/probe/$build/Cargo.toml")"
  (cd "$tree" && sha256sum crates/pert-ui-runtime/src/text.rs crates/pert-ui-runtime/src/text/tabs.rs crates/pert-ui-runtime/src/text_fields.rs crates/pert-ui-runtime/src/text_selection.rs)
  (cd "$E/probe/$build" && RUSTUP_TOOLCHAIN=1.99.1 CARGO_TARGET_DIR=$target cargo run --offline -q --release 2>&1)
  echo "exit status: $?"
  echo "finished_at_utc: $(date -u +%FT%TZ)"
} > "$out" 2>&1
tail -n 2 "$out"
